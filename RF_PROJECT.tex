% Owner of the block. Empty argument prints "not assigned".
\newcommand{\blockowner}[1]{%
	\vspace{-0.5\baselineskip}%
	\noindent{\small\textit{Owner: %
	\if\relax\detokenize{#1}\relax not assigned\else#1\fi}}\par
	\vspace{0.5\baselineskip}}

% Reference to a section of the design proposal, e.g. \dpref{5} or \dpref{8, I2}.
\newcommand{\dpref}[1]{DP~\S#1}
% Reference to the project specification (projekt.md), e.g. \specref{3}.

% Inline placeholder for text that is still to be written.
\newcommand{\placeholder}[1]{\textit{[TODO: #1]}}

% Figure placeholder. Replace with a real figure once the image exists:

% Requirements table.
%   \req[status]{ID}{Parameter}{Requirement}{Reference}
%   status: open (default) | met | not met | waived
\newenvironment{requirements}{%
	\par\medskip\noindent\small
	\tabularx{\textwidth}{@{}l P{0.27\textwidth} L l l@{}}
	\toprule
	\textbf{ID} & \textbf{Parameter} & \textbf{Requirement} & \textbf{Ref.} & \textbf{Status} \\
	\midrule}{%
	\bottomrule
	\endtabularx\par\normalsize}
\newcommand{\req}[5][open]{#2 & #3 & #4 & #5 & #1 \\}

% Changelog (register of decisions and changes).
%   \change{Date}{Author}{What (was -> is)}{Why}{Affected blocks}{Status}

	\par\medskip\noindent\small
	\tabularx{\textwidth}{@{}l P{0.13\textwidth} L P{0.2\textwidth} P{0.12\textwidth} l@{}}
	\toprule
	\textbf{Date} & \textbf{Author} & \textbf{What (was $\to$ is)} & \textbf{Why} & \textbf{Affects} & \textbf{Status} \\
	\midrule}{%
	\bottomrule
	\endtabularx\par\normalsize}
