\section{Requirements}
Source: project specification and requirements changeover v0.2.

\req{LO-01}{VDD}{$1.1\,\mathrm{V}$}{SYS-04}
	\req{LO-02}{Input interface}{Control voltage $V_{\mathrm{tune}}$: $0$--$1.1\,\mathrm{V}$ DC + digital capacitor-bank control bits $SW\langle N{-}1{:}0\rangle$ ($0$ / $1.1\,\mathrm{V}$)}{}
	\req{LO-03}{Input coupling}{DC (varactor tuning \& switched-capacitor bank)}{}
	\req{LO-04}{Input DC voltage}{$0.55\,\mathrm{V}$ nominal for $V_{\mathrm{tune}}$ ($V_{\mathrm{DD}}/2$)}{}
	\req{LO-05}{Input signal}{N/A (autonomous oscillator)}{}
	\req{LO-06}{Input impedance}{$> 10\,\mathrm{k\Omega}$ at $V_{\mathrm{tune}}$ (high-impedance varactor/bias node)}{}
	\req{LO-07}{Output interface}{Quadrature I/Q (two differential pairs: $I_{+}/I_{-}$ and $Q_{+}/Q_{-}$, $90^\circ$ phase shift), I3 and I7 together}{I3, I7}
	\req{LO-08}{Output coupling}{AC: coupling capacitor + bias network at the mixers (DMIX, UMIX)}{I3, I7, SYS-11}
	\req{LO-09}{Output DC voltage}{$\approx 0.55\,\mathrm{V}$ (buffer common-mode bias, blocked by AC coupling)}{I3, I7}
	\req{LO-10}{Output signal}{$0.6\,\mathrm{V_{pp,diff}}$ per I/Q branch ($0.4$--$0.8\,\mathrm{V_{pp,diff}}$); internal VCO core swing $1.0$--$1.3\,\mathrm{V_{pp,se}}$}{I3, I7, DMIX-20}
	\req{LO-11}{Output impedance / load}{Per I/Q branch: $(R_{\mathrm{DMIX}} \parallel R_{\mathrm{UMIX}}) \parallel (C_{\mathrm{DMIX}} + C_{\mathrm{UMIX}})$; target $R_L \geq 2\,\mathrm{k\Omega} \parallel C_L \leq 250\,\mathrm{fF}$ diff.}{I3, I7, DMIX-21}
	\req{LO-12}{Frequency / bandwidth}{$6.0\,\mathrm{GHz}$ nominal (Wi-Fi\,6E $6\,\mathrm{GHz}$ band); output tuning range $5.7$--$6.45\,\mathrm{GHz}$ (internal LC-VCO core: $11.4$--$12.9\,\mathrm{GHz}$ plus PVT margin)}{SYS-02}
	\req{LO-13}{Gain}{VCO tuning sensitivity $K_{\mathrm{VCO}} \approx 50$--$150\,\mathrm{MHz/V}$ at $6\,\mathrm{GHz}$ output ($100$--$300\,\mathrm{MHz/V}$ at $12\,\mathrm{GHz}$ core)}{}
	\req{LO-14}{Noise}{N/A (specified via Phase Noise in LO-22)}{}
	\req{LO-15}{Linearity}{N/A (saturating/limiting LO path; harmonic suppression $> 20\,\mathrm{dBc}$ at buffer output)}{}
	\req{LO-16}{Supply current}{$\leq 13\,\mathrm{mA}$ total ($12\,\mathrm{GHz}$ LC-VCO core $\approx 4$--$5\,\mathrm{mA}$ + $\div 2$ CML divider $\approx 3$--$4\,\mathrm{mA}$ + I/Q buffers $\approx 4$--$6\,\mathrm{mA}$)}{\specref{3}}
	\req{LO-17}{Power consumption}{$\leq 14.3\,\mathrm{mW}$ at $1.1\,\mathrm{V}$}{SYS-04}
	\req{LO-22}{Phase noise}{$\leq -110\,\mathrm{dBc/Hz}$ at $1\,\mathrm{MHz}$ offset at $6\,\mathrm{GHz}$ output ($\leq -104\,\mathrm{dBc/Hz}$ at $12\,\mathrm{GHz}$ VCO core for Wi-Fi\,6 1024-QAM); I/Q phase error $< 1^\circ$, amplitude mismatch $< 0.2\,\mathrm{dB}$}{SYS-02, SYS-06}
\end{requirements}
