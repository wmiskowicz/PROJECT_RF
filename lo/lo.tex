\section{Requirements}
Source: project specification and requirements changeover v0.2.

\begin{requirements}
	\req{LO-01}{VDD}{1.1\,V}{SYS-04}
	\req{LO-02}{Input interface}{N/A ($V_{\mathrm{tune}}$: $0$--$1.1\,\mathrm{V}$ DC)}{}
	\req{LO-03}{Input coupling}{N/A (DC for $V_{\mathrm{tune}}$)}{}
	\req{LO-04}{Input DC voltage}{N/A ($0.55\,\mathrm{V}$ nominal for $V_{\mathrm{tune}}$)}{}
	\req{LO-05}{Input signal}{N/A}{}
	\req{LO-06}{Input impedance}{N/A ($>10\,\mathrm{k\Omega}$ at $V_{\mathrm{tune}}$)}{}
	\req{LO-07}{Output interface}{quadrature I/Q (two diff.\ pairs: I and Q, $90^\circ$ shift), I3 and I7 together}{I3, I7}
	\req{LO-08}{Output coupling}{AC: C + R at the mixers (DMIX, UMIX)}{I3, I7, SYS-11}
	\req{LO-09}{Output DC voltage}{$\approx 0.55\,\mathrm{V}$ (buffer bias, blocked by AC coupling)}{I3, I7}
	\req{LO-10}{Output signal}{$0.6\,\mathrm{V_{pp,diff}}$ per I/Q branch ($0.4$--$0.8\,\mathrm{V_{pp,diff}}$)}{I3, I7, DMIX-20}
	\req{LO-11}{Output impedance / load}{per I/Q branch: $(R_{\mathrm{DMIX}} \parallel R_{\mathrm{UMIX}}) \parallel (C_{\mathrm{DMIX}} + C_{\mathrm{UMIX}})$; target $R_L \geq 2\,\mathrm{k\Omega} \parallel C_L \leq 250\,\mathrm{fF}$ diff.}{I3, I7, DMIX-21}
	\req{LO-12}{Frequency / bandwidth}{$6.0\,\mathrm{GHz}$ nominal (Wi-Fi\,6E 6\,GHz band); tuning range $5.7$--$6.45\,\mathrm{GHz}$}{SYS-02}
	\req{LO-13}{Gain}{N/A}{}
	\req{LO-14}{Noise}{N/A}{}
	\req{LO-15}{Linearity}{N/A}{}
	\req{LO-16}{Supply current}{$\leq 13\,\mathrm{mA}$ (12\,GHz VCO $\approx 4\,\mathrm{mA}$ + $\div 2$ $\approx 3\,\mathrm{mA}$ + I/Q buffers $\approx 6\,\mathrm{mA}$)}{\specref{3}}
	\req{LO-17}{Power consumption}{$\leq 14.3\,\mathrm{mW}$ at $1.1\,\mathrm{V}$}{SYS-04}
	\req{LO-22}{Phase noise}{$\leq -110\,\mathrm{dBc/Hz}$ at $1\,\mathrm{MHz}$ offset (Wi-Fi\,6 1024-QAM); I/Q phase error $< 1^\circ$}{SYS-02, SYS-06}
\end{requirements}
