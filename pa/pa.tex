\section{Requirements}
Source: design proposal v0.1~\cite{design_proposal} and project
specification~\cite{project_spec}. Values become binding after approval at
meeting~2; every change goes through the changelog below.

\begin{requirements}
	\req{PA-01}{Gain}{$+15$\,dB (input $+1$\,dBm)}{\dpref{6}}
	\req{PA-02}{Output power}{$+16$\,dBm average into 50\,$\Omega$ (useful sideband)}{\dpref{6}}
	\req{PA-03}{Output compression}{OP1dB $\geq +24$\,dBm (8\,dB back-off)}{\dpref{6}, \specref{3}}
	\req{PA-04}{Input from DRV (I11)}{differential RF, AC-coupled; $C_\mathrm{in} \leq 500$\,fF diff.; own bias; matching agreed jointly with DRV}{\dpref{8, I11}}
	\req{PA-05}{Output to antenna (I12)}{differential to single-ended 50\,$\Omega$ via balun; load 50\,$\Omega$ + pad 100\,fF; 1.41\,Vrms (peak 2.0\,V), 5.0\,V peak at P1dB}{\dpref{8, I12}}
	\req{PA-06}{2nd harmonic}{H2 (12\,GHz) $\leq -30$\,dBc}{\dpref{6}}
	\req{PA-07}{Supply and power}{VDDPA 2\,V $\pm5\%$, $\leq 350$\,mW (175\,mA); voltage stress of transistors checked}{\dpref{9}, \specref{3}}
	\req{PA-08}{Efficiency}{PAE $\geq 10\%$ at $+16$\,dBm}{\dpref{6}}
	\req{PA-09}{Two-tone test}{6.000 and 6.001\,GHz, $+13$\,dBm per tone}{\dpref{10}}
	\req{PA-10}{Modulated-signal test}{PAPR $\approx 10$\,dB at $+16$\,dBm; report EVM and ACPR; EVM $\leq -25$\,dB}{\dpref{6}, \specref{3}}
	\req{PA-11}{Bias}{internal bias circuit, real \texttt{tsmcN40} devices (transformers/baluns from shared model)}{\specref{4}}
\end{requirements}
