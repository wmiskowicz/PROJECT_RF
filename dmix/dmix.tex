\section{Requirements}
Source: design proposal v0.1~\cite{design_proposal} and project
specification~\cite{project_spec}. Values become binding after approval at
meeting~2; every change goes through the changelog below.

\begin{requirements}
	\req{DMIX-01}{Conversion gain}{$+8$\,dB}{\dpref{5}}
	\req{DMIX-02}{Noise figure}{$\leq 15$\,dB (SSB)}{\dpref{5}}
	\req{DMIX-03}{IIP3}{$\geq -5$\,dBm (at RF input)}{\dpref{5}}
	\req{DMIX-04}{RF input from LNA (I2)}{differential via transformer; $C_\mathrm{in} \leq 80$\,fF diff.; 1.8--7.1\,mVrms diff.}{\dpref{8, I2}}
	\req{DMIX-05}{LO input (I3)}{differential, AC-coupled, 0.6\,V peak diff.; $C_\mathrm{in} \leq 80$\,fF diff.; own bias}{\dpref{8, I3}}
	\req{DMIX-06}{BB output to VGA (I4)}{differential, AC-coupled, $f_\mathrm{c} \leq 100$\,kHz; 4.5--18\,mVrms diff.; $Z_\mathrm{out} \leq 2$\,k$\Omega$ diff.}{\dpref{8, I4}}
	\req{DMIX-07}{Supply and current}{1.1\,V, 7\,mA (8\,mW)}{\dpref{9}}
	\req{DMIX-08}{Bias}{internal bias circuit, real \texttt{tsmcN40} devices only}{\specref{4}}
\end{requirements}

\noindent\textit{Note:} the LPF is assigned to the VGA designer in the design
proposal (decision~7, \dpref{13}); by default in the specification it belongs
to DMIX.
