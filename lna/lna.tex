\section{Requirements}
Source: project specification and requirements changeover v0.2.

\begin{requirements}
	\req{LNA-01}{VDD}{1.1\,V}{SYS-04}
	\req{LNA-02}{Input interface}{single-ended, I1}{I1}
	\req{LNA-03}{Input coupling}{AC: series C in the input match (LNA)}{I1}
	\req{LNA-04}{Input DC voltage}{internal (gate bias through a resistor), TBD, for information only}{I1}
	\req{LNA-05}{Input signal}{$-82$\dots$-30$\,dBm ($\approx 50\,\mu$Vpp\dots20\,mVpp at the matched pin, SE)}{SYS-07}
	\req{LNA-06}{Input impedance}{50\,$\Omega$, $|S_{11}| \leq -10$\,dB, 5.9--6.1\,GHz}{I1}
	\req{LNA-07}{Output interface}{differential, I2}{I2}
	\req{LNA-08}{Output coupling}{AC: transformer used as the LNA load (LNA)}{I2}
	\req{LNA-09}{Output DC voltage}{centre tap, set by the DMIX bias requirement (TBD with DMIX)}{I2}
	\req{LNA-10}{Output signal}{$\approx 0.28$\,mVpp\dots112\,mVpp diff.\ (15\,dB above the input)}{I2}
	\req{LNA-11}{Output impedance / load}{DMIX input (DMIX-06), TBD; earlier estimate $C \leq 80$\,fF diff.}{I2}
	\req{LNA-12}{Frequency / bandwidth}{6\,GHz $\pm$ 10\,MHz}{SYS-02}
	\req{LNA-13}{Gain}{\textbf{15\,dB} voltage gain (single gain mode)}{SYS-06}
	\req{LNA-14}{Noise}{NF $\leq$ 2\,dB}{SYS-06}
	\req{LNA-15}{Linearity}{IIP3 TBD (integrator budget); the chain must reach $P_\mathrm{1dB,in} \geq -22$\,dBm ($-30$\,dBm + 8\,dB back-off), so the LNA needs at least this headroom}{SYS-07}
	\req{LNA-16}{Supply current}{TBD}{}
	\req{LNA-17}{Power consumption}{TBD}{}
\end{requirements}
