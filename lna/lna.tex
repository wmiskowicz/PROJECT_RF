\section{Requirements}
Source: design proposal v0.1~\cite{design_proposal} and project
specification~\cite{project_spec}. Values become binding after approval at
meeting~2; every change goes through the changelog below.

\begin{requirements}
	\req{LNA-01}{Gain, HG mode}{$+20$\,dB}{\dpref{4}}
	\req{LNA-02}{Gain, LG mode}{0\,dB; selected by \texttt{lna\_lg} = 1 (1.1\,V logic, static)}{\dpref{4}, \dpref{8, S3}}
	\req{LNA-03}{Noise figure, HG mode}{$\leq 4.5$\,dB}{\dpref{5}}
	\req{LNA-04}{Noise figure, LG mode}{$\leq 7$\,dB}{\dpref{5}}
	\req{LNA-05}{IIP3, LG mode}{$\geq +5$\,dBm}{\dpref{5}}
	\req{LNA-06}{Input match (I1)}{single-ended 50\,$\Omega$, $|S_{11}| \leq -10$\,dB in 5.9--6.1\,GHz; series capacitor inside LNA; pad + ESD 100\,fF in testbench}{\dpref{8, I1}}
	\req{LNA-07}{Input signal level (I1)}{$-82$\dots$-30$\,dBm, peak up to $-20$\,dBm}{\dpref{8, I1}}
	\req{LNA-08}{Output to DMIX (I2)}{differential via transformer (no shared DC); 1.8--7.1\,mVrms diff.; load $C_\mathrm{in} \leq 80$\,fF diff.; common transformer model}{\dpref{8, I2}}
	\req{LNA-09}{Supply and current}{1.1\,V, 10\,mA (11\,mW)}{\dpref{9}}
	\req{LNA-10}{Bias}{internal bias circuit, real \texttt{tsmcN40} devices only}{\specref{4}}
	\req{LNA-11}{Modulated-signal test}{required for LNA; shared test signal, PAPR $\approx 10$\,dB}{\specref{3}}
\end{requirements}
