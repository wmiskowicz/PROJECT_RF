\section{Requirements}
Source: project specification, requirements changeover v0.2 and the system budget v0.4
(section~\ref{sec:integrator-budget}).

\begin{requirements}
	\req{DRV-01}{VDD}{1.1\,V}{SYS-04}
	\req{DRV-02}{Input interface}{differential, I8}{I8}
	\req{DRV-03}{Input coupling}{AC: transformer (UMIX load), see UMIX-08}{I8}
	\req{DRV-04}{Input DC voltage}{internal, for information: gate bias $\approx 0.5$\,V set by the DRV through $\geq 5$\,k$\Omega$ per side (UMIX secondary DC-floating)}{I8}
	\req{DRV-05}{Input signal}{0.1\,V pk diff.\ nominal (UMIX-10); 0.32\,V pk diff.\ when the DRV is at its $OP_\mathrm{1dB}$}{I8}
	\req{DRV-06}{Input impedance}{$R_\mathrm{in} \geq 1$\,k$\Omega$ $\parallel$ $C_\mathrm{in} \leq 150$\,fF diff.\ at 6\,GHz}{I8}
	\req{DRV-07}{Output interface}{differential, I9}{I9}
	\req{DRV-08}{Output coupling}{AC: transformer used as the DRV load (DRV); resonates the PA input capacitance}{I9}
	\req{DRV-09}{Output DC voltage}{primary centre tap at VDD (drain peaks up to $\approx 2\,$VDD: cascode or thick-oxide device); secondary DC-floating, the PA biases its own gates (PA-04)}{I9}
	\req{DRV-10}{Output signal}{0.5\,V pk diff.\ (1.0\,Vpp) at the PA gates for $+16$\,dBm at the antenna}{I9, BUD-06}
	\req{DRV-11}{Output impedance / load}{PA input $R \geq 100\,\Omega$ $\parallel$ $C \leq 1.0$\,pF diff.\ (PA-06)}{I9}
	\req{DRV-12}{Frequency / bandwidth}{6\,GHz $\pm$ 10\,MHz; gain flatness $\leq 0.5$\,dB in 5.99--6.01\,GHz}{SYS-02}
	\req{DRV-13}{Gain}{\textbf{14\,dB $\pm$ 1.5\,dB} voltage gain (PA gate diff.\ / DRV input diff.), transformer included}{BUD-06}
	\req{DRV-14}{Noise}{N/A}{}
	\req{DRV-15}{Linearity}{$OP_\mathrm{1dB} \geq 1.6$\,V pk diff.\ at the PA gates (10\,dB above nominal, 1\,dB above the PA input at the PA $P_\mathrm{1dB}$)}{BUD-05}
	\req{DRV-16}{Supply current}{$\leq 25$\,mA (budget)}{BUD-09}
	\req{DRV-17}{Power consumption}{$\leq 27.5$\,mW}{BUD-09}
\end{requirements}

\noindent\textit{Budget notes.} The PA input needs 1.41\,V pk diff.\ at the PA
$P_\mathrm{1dB}$. A 1:1.5 to 1:2 step-up transformer lets the DRV meet DRV-15 with
$\leq 0.55$\,V pk per side on its own drains at 1.1\,V.
