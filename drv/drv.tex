\section{Requirements}
Source: design proposal v0.1~\cite{design_proposal} and project
specification~\cite{project_spec}. Values become binding after approval at
meeting~2; every change goes through the changelog below.

\begin{requirements}
	\req{DRV-01}{Gain}{$+18$\,dB}{\dpref{6}}
	\req{DRV-02}{Signal levels}{input $-17$\,dBm, output $+1$\,dBm average}{\dpref{6}}
	\req{DRV-03}{Output compression}{OP1dB $\geq +10$\,dBm (1\,dB above PA IP1dB)}{\dpref{6}}
	\req{DRV-04}{Input from BPF (I10)}{differential RF, AC-coupled; $C_\mathrm{in} \leq 100$\,fF diff.; own bias}{\dpref{8, I10}}
	\req{DRV-05}{Output to PA (I11)}{differential RF, AC-coupled; $+1$\,dBm (0.36\,Vrms diff., peak $\approx 1.1$\,V); load $C_\mathrm{in,PA} \leq 500$\,fF diff.; matching network agreed jointly with PA}{\dpref{8, I11}}
	\req{DRV-06}{Supply and current}{1.1\,V (decision~6), 41\,mA (45\,mW)}{\dpref{9}, \dpref{13}}
	\req{DRV-07}{Bias}{internal bias circuit, real \texttt{tsmcN40} devices only}{\specref{4}}
\end{requirements}
